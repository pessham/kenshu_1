function c(l, t) {
  let x = 0;
  // dはdataのつもりらしい
  for (let i = 0; i < l.length; i++) {
    let p = l[i].p;
    let q = l[i].q;
    if (l[i].f) {
      x += (p * 0.9) * q;
    } else {
      x += p * q;
    }
  }
  if (t) {
    x = x * 1.1;
  }
  return Math.floor(x);
}
